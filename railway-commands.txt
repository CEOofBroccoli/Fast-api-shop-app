# Railway Quick Start Commands

# 1. Install Railway CLI (Windows)
# Download from: https://railway.app/cli
# Or using npm:
npm install -g @railway/cli

# 2. Login to Railway
railway login

# 3. Initialize Railway project in your directory
railway init

# 4. Add PostgreSQL service
railway add postgresql

# 5. Add Redis service (optional)
railway add redis

# 6. Set environment variables (use values from .env.railway)
railway variables set JWT_SECRET_KEY="your-secret-key-here"
railway variables set ENVIRONMENT="production"
railway variables set DEBUG="false"

# 7. Deploy your application
railway up

# 8. Check deployment status
railway status

# 9. View logs
railway logs

# 10. Open your deployed app
railway open
